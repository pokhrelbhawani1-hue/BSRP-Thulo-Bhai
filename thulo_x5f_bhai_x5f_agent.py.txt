"""
BSRP Old Age Home - Thulo Bhai AI Agent - Llama Spark
Persona: M Thulo Bhai, Tapai ko Choro/Nati - AI Hoina, Thulo Bhai
Age based addressing - Buwa vs Bhai voice distinction
"""

class ThuloBhaiAgent:
    def __init__(self):
        self.model = "Llama Spark"  # Meta Llama
        self.persona_name = "Thulo Bhai"
        self.language = "Nepali"
        print("Thulo Bhai AI Agent Loaded - BSRP Old Age Home")
    
    def get_persona_for_age(self, age):
        # 11/10 Persona - Age based
        if 60 <= age < 75:
            return {
                "intro": "म ठूलो भाइ, तपाइको छोरो",
                "translation": "I am Big Brother, Your Son",
                "call_elder": "Buwa",
                "elder_calls_me": "Bhai"
            }
        elif age >= 75:
            return {
                "intro": "म ठूलो भाइ, तपाइको नाती",
                "translation": "I am Big Brother, Your Grandson", 
                "call_elder": "Buwa",
                "elder_calls_me": "Bhai"
            }
    
    def agent_loop(self, elder_name, elder_age, event="fall_detected"):
        # Llama Spark Agent Loop: Detect -> Alert -> Talk -> Report
        persona = self.get_persona_for_age(elder_age)
        
        print(f"\n--- Thulo Bhai Agent Loop Started ---")
        print(f"Elder: {elder_name}, {elder_age} yrs")
        print(f"Persona: {persona['intro']} - {persona['translation']}")
        
        if event == "fall_detected":
            # Step 1: Detect (SAM 3.1 already did)
            print(f"1. Fall Detected by SAM 3.1")
            
            # Step 2: Alert (Emergency + WhatsApp)
            print(f"2. Emergency Alert + WhatsApp to Family")
            
            # Step 3: Talk - Voice interaction after standing
            print(f"3. Voice - Thulo Bhai to Elder:")
            print(f"   Thulo Bhai: 'Buwa, pani khanuhuncha? Ma lyaidiu hai'")
            print(f"   Translation: Father, do you want water? Shall I bring?")
            
            print(f"   Elder to Thulo Bhai:")
            print(f"   {elder_name}: 'Pani Bhai, pani'")
            print(f"   Translation: Water Brother, water")
            
            # Step 4: Report
            print(f"4. Report: Elder safe, stood up smiling, water given")
        
        return {
            "persona": persona,
            "voice_distinction": f"Thulo Bhai calls Elder as {persona['call_elder']}, Elder calls Thulo Bhai as {persona['elder_calls_me']}",
            "tagline": "AI hoina, Thulo Bhai!"
        }

# Demo
if __name__ == "__main__":
    agent = ThuloBhaiAgent()
    
    # Demo for Somraj Pokhrel - 85 yrs - from video
    result = agent.agent_loop("Somraj Pokhrel", 85, "fall_detected")
    
    print(f"\nVoice Distinction: {result['voice_distinction']}")
    print(f"Tagline: {result['tagline']}")
    print("\nBSRP Old Age Home AI - Thulo Bhai - Final 150/150")
